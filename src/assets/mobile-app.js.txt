function token(){
  const q = new URLSearchParams(location.search).get('token');
  if (q) { try { localStorage.setItem('nyra-token', q); } catch(e){} return q; }

async function api(path, opts){
  opts = opts || {};
  const res = await fetch(path, { ...opts, headers: { 'X-Nyra-Token': token(), 'Content-Type': 'application/json' } });

  if (res.status === 403) { lostDevice(); throw new Error((data && data.error) || t('m.unauthorized')); }
  if (!res.ok) throw new Error((data && data.error) || ('HTTP ' + res.status));
  return data;
}

function deviceName(){
  const ua = navigator.userAgent || '';
  if (/iPhone/.test(ua)) return 'iPhone';
  if (/iPad/.test(ua)) return 'iPad';
  const m = /Android[^;]*;\s*([^;)]+?)(?:\sBuild|\))/.exec(ua);
  if (m) return m[1].trim().slice(0, 40);
  if (/Android/.test(ua)) return 'Android';
  return 'Phone';
}

function relTime(iso){
  if (!iso) return '';
  const d = new Date(/Z|[+-]\d\d:?\d\d$/.test(iso) ? iso : iso + 'Z');
  const s = Math.max(0, (Date.now() - d.getTime()) / 1000);
  if (!isFinite(s) || s < 60) return t('m.time.now');
  if (s < 3600) return t('m.time.m', { n: Math.floor(s / 60) });
  if (s < 86400) return t('m.time.h', { n: Math.floor(s / 3600) });
  return t('m.time.d', { n: Math.floor(s / 86400) });
}
function threadState(s){
  if (lastApprovals.some(a => a.session === s.id)) return ['wait', t('m.st.approval')];
  if (s.status === 'running') return ['run', t('m.st.running')];
  if (s.status === 'error') return ['err', t('m.st.error')];
  const seen = seenMap()[s.id];
  if (seen !== undefined && (s.count || 0) > seen) return ['review', t('m.st.review')];
  return ['', t('m.st.idle', { n: s.count || 0 })];
}

function textOf(m){
  if (typeof m.content === 'string') return m.content;
  if (Array.isArray(m.content)) return m.content.map(p => (p && p.text) || '').join('');
  return m.content == null ? '' : JSON.stringify(m.content);
}
function stepOf(c){
  const name = (c.function && c.function.name) || '?';
  let args = {}; try { args = JSON.parse((c.function && c.function.arguments) || '{}'); } catch(e){}
  const line = el('span');
  const put = (verb, code) => { line.appendChild(document.createTextNode(t(verb) + ' ')); line.appendChild(el('code', '', code)); };
  if (name === 'shell' && args.command) put('m.step.ran', String(args.command));
  else if (name === 'read_file' && args.path) put('m.step.read', String(args.path));
  else if (EDIT_TOOLS.includes(name) && args.path) put('m.step.edited', String(args.path));
  else put('m.step.called', name + (args.path ? ' ' + args.path : ''));
  return line;
}

// 运行中的这一轮还没写进会话（网关只给出本轮之前的消息）：从事件流补上本轮的用户消息、工具步骤和中间回复。
function liveTurn(v){
  const out = [];
  if (v.status !== 'running' || !Array.isArray(v.events)) return out;
  let last = -1; v.events.forEach((e, i) => { if (e.type === 'user') last = i; });
  if (last < 0) return out;
  v.events.slice(last).forEach(e => {
    const d = e.data || {};
    if (e.type === 'user') out.push({ role: 'user', content: String(d.content || '') });
    else if (e.type === 'tool_start') out.push({ role: 'assistant', content: '', tool_calls: [{ function: { name: String(d.name || '?'), arguments: typeof d.arguments === 'string' ? d.arguments : JSON.stringify(d.arguments || {}) } }] });
    else if (e.type === 'assistant') out.push({ role: 'assistant', content: String(d.content || '') });
  });
  return out;
}
function renderFeed(v){
  const all = v.messages || [];
  const msgs = all.filter(m => m.role !== 'system').concat(liveTurn(v));
  const key = msgs.length + ':' + (msgs.length ? textOf(msgs[msgs.length - 1]).length : 0) + ':' + v.status + ':' + (v.next || 0);
  $('chat-status').textContent = v.status === 'running' ? t('m.working') : (v.status === 'error' ? t('m.errorPrefix', { msg: v.error || '' }) : '');
  const firstUser = msgs.find(m => m.role === 'user');
  if (firstUser) $('chat-title').textContent = textOf(firstUser).slice(0, 60) || t('m.untitled');
  if (v.status !== 'running') markSeen(currentSession, all.length);
  if (key === lastFeedKey) return;
  lastFeedKey = key;
  const feed = $('feed'); feed.innerHTML = '';
  let steps = [];
  const flush = (open) => { if (steps.length) { feed.appendChild(stepsBlock(steps, open)); steps = []; } };
  msgs.forEach(m => {
    if (m.role === 'tool') return;
    const txt = textOf(m);
    if (m.role === 'assistant' && m.tool_calls && m.tool_calls.length) {
      if (txt) { flush(false); feed.appendChild(bubble(m, all, txt)); }
      m.tool_calls.forEach(c => steps.push(stepOf(c)));
      return;
    }
    if (!txt) return;
    flush(false);
    feed.appendChild(bubble(m, all, txt));
  });
  flush(false);
  // 运行中的这一轮还没写进会话：先把刚发出的消息显示出来。
  if (pendingText && v.status === 'running' && !msgs.some(m => m.role === 'user' && textOf(m) === pendingText)) feed.appendChild(el('div', 'u', pendingText));
  if (v.status !== 'running') pendingText = '';
  window.scrollTo(0, document.body.scrollHeight);
}
function bubble(m, all, txt){
  const b = el('div', m.role === 'user' ? 'u' : 'p', txt);
  // 点一下消息展开回退操作（与桌面页一致：回退到此 / 编辑重发 / Fork 新会话）。
  const index = all.indexOf(m);
  if (index >= 0) b.onclick = () => toggleRewind(b, index, txt);
  return b;
}
function toggleRewind(b, index, text){
  const next = b.nextSibling;
  if (next && next.classList && next.classList.contains('rw')) { next.remove(); return; }
  const row = el('div', 'rw');
  if (b.classList.contains('u')) row.style.justifyContent = 'flex-end';
  [['m.rewind.truncate', 'truncate'], ['m.rewind.edit', 'edit'], ['m.rewind.fork', 'fork']].forEach(([label, action]) => {
    const x = el('button', '', t(label)); x.onclick = () => doRewind(index, action, text); row.appendChild(x);
  });
  b.after(row);
}
async function doRewind(index, action, text){
  if (!currentSession) return;
  const rows = await api('/api/sessions/' + encodeURIComponent(currentSession) + '/history').catch(() => null);
  const messageId = rows && rows.messages && rows.messages[index] ? rows.messages[index].id : null;
  if (messageId === null) { toast(t('m.rewind.notYet')); return; }
  let edited = null;
  if (action === 'edit') { edited = prompt(t('m.rewind.editPrompt'), text || ''); if (edited === null) return; }
  try {
    const res = await api('/api/sessions/' + encodeURIComponent(currentSession) + '/rewind', { method: 'POST', body: JSON.stringify({ message_id: messageId, action, text: edited }) });
    if (res && res.new_session_id) { await openSession(res.new_session_id); return; }
    lastFeedKey = ''; await tick();
  } catch (e) { toast(t('m.rewind.failed', { msg: e.message })); }
}

function ruleLabel(sg){
  if (!sg) return '';
  if (sg.kind === 'prefix' && Array.isArray(sg.prefix)) return t('m.rule.prefix', { prefix: sg.prefix.join(' ') });
  if (sg.kind === 'edits') return t('m.rule.edits');
  return sg.label || '';
}
function parseDiff(summary){
  const lines = String(summary || '').split('\n');
  let path = '', add = 0, del = 0; const body = [];
  lines.slice(1).forEach(l => {
    if (l.startsWith('+++ ')) { path = l.replace(/^\+\+\+ (b\/)?/, ''); return; }
    if (l.startsWith('--- ')) { if (!path) path = l.replace(/^--- (a\/)?/, ''); return; }
    if (l.startsWith('+')) { add++; body.push(['a', '+', l.slice(1)]); }
    else if (l.startsWith('-')) { del++; body.push(['d', '−', l.slice(1)]); }
    else if (l.trim()) body.push(['', '', l]);
  });
  return { path, add, del, body, head: lines[0] || '' };
}

async function decide(id, body){
  try {
    await api('/api/approvals/' + encodeURIComponent(id), { method: 'POST', body: JSON.stringify(body) });
    delete cardState[id]; sheetKey = '';
    lastApprovals = lastApprovals.filter(a => a.id !== id);
    toast(body.approve ? (body.scope ? t('m.toast.remembered') : t('m.toast.allowed')) : (body.feedback ? t('m.toast.deniedFeedback') : t('m.toast.denied')));
    renderSheet();
    await tick();
  } catch (e) { toast(t('m.toast.failed', { msg: e.message })); }
}
async function tick(){
  try {

    lastApprovals = data.approvals || [];
    document.title = lastApprovals.length ? '(' + lastApprovals.length + ') Nyra' : 'Nyra';

    else if (currentTab === 'threads' && !$('threads-view').classList.contains('hidden')) await loadSessions();
  } catch (e) { /* 断网时下次再试 */ }
}

function scheduleText(task){

  const s = Number(task.every_seconds || 0);
  if (s >= 3600 && s % 3600 === 0) return t('m.auto.everyHour', { n: s / 3600 });

}

    const tx = el('div', 'tx'); tx.appendChild(el('b', '', task.name));
    tx.appendChild(el('small', '', on ? scheduleText(task) + (task.mode ? ' · ' + t('m.mode.' + task.mode) : '') : t('m.auto.paused')));
    row.appendChild(tx);
    const tg = el('button', 'tg' + (on ? '' : ' off')); tg.dataset.act = 'toggle'; tg.setAttribute('role', 'switch'); tg.setAttribute('aria-checked', String(on)); tg.setAttribute('aria-label', task.name);
    tg.onclick = async () => {
      tg.disabled = true;
      try { const r = await api('/api/schedule', { method: 'POST', body: JSON.stringify({ action: 'toggle', name: task.name }) }); toast(t(r.task && r.task.enabled === false ? 'm.auto.off' : 'm.auto.on', { name: task.name })); }
      catch (e) { toast(t('m.auto.toggleFailed', { msg: e.message })); }
      await loadAutomations();
    };
    row.appendChild(tg);
    box.appendChild(row);
  });
  const ran = list.filter(x => x.last_run).sort((a, b) => String(b.last_run).localeCompare(String(a.last_run))).slice(0, 3);
  if (ran.length) {
    box.appendChild(el('div', 'lh', t('m.auto.lastRun')));
    ran.forEach(task => {
      const row = el('div', 'it');
      const ok = task.last_status === 'ok';
      row.appendChild(el('span', ok ? 'ok' : 'bad', ok ? '✓' : '✕'));
      const tx = el('div', 'tx'); tx.appendChild(el('b', '', task.name));
      tx.appendChild(el('small', 'wrap', fmtWhen(task.last_run) + (task.last_summary ? ' · ' + task.last_summary : '')));
      row.appendChild(tx); box.appendChild(row);
    });
  }
}

function renderModes(){
  const box = $('modes'); box.innerHTML = '';
  const cur = lastInfo ? lastInfo.mode : '';
  const modes = ['readonly', 'auto', 'full'];
  if (cur === 'creator') modes.push('creator');
  modes.forEach(m => {
    const b = el('button', m === cur ? 'on' : '', modeLabel(m)); b.dataset.mode = m;
    b.onclick = () => chooseMode(m);
    box.appendChild(b);
  });
  const c = $('mode-confirm'); c.innerHTML = '';
  if (fullStep === 1) {
    const w = el('div', 'warn');
    w.appendChild(el('b', '', t('m.full.warnTitle')));
    w.appendChild(el('div', 'note', t('m.full.warnBody')));
    const go = el('button', 'bt danger', t('m.full.continue')); go.dataset.act = 'full-step1'; go.onclick = () => { fullStep = 2; renderModes(); };
    const no = el('button', 'bt ghost', t('m.cancel')); no.onclick = () => { fullStep = 0; renderModes(); };
    w.appendChild(go); w.appendChild(no); c.appendChild(w);
  } else if (fullStep === 2) {
    const w = el('div', 'warn');
    w.appendChild(el('b', '', t('m.full.confirmTitle')));
    const go = el('button', 'bt danger', t('m.full.confirm')); go.dataset.act = 'full-step2'; go.onclick = () => setMode('full');
    const no = el('button', 'bt ghost', t('m.cancel')); no.onclick = () => { fullStep = 0; renderModes(); };
    w.appendChild(go); w.appendChild(no); c.appendChild(w);
  }
}
function chooseMode(m){
  if (lastInfo && m === lastInfo.mode) return;
  if (m === 'full') { fullStep = 1; renderModes(); return; }
  setMode(m);
}

// ---------------------------------------------------------------- 字体 ---
// CSP 不放开 font-src：同源 fetch 拿到 woff 字节，再用 FontFace 从 ArrayBuffer 加载。
function loadFonts(){
  if (!window.FontFace || !document.fonts) return;
  [['Inter', 'inter-400', '400'], ['Inter', 'inter-600', '500 700'], ['JetBrains Mono', 'jbm-400', '400']].forEach(([fam, file, weight]) => {
    fetch('/m/fonts/' + file + '.woff').then(r => r.ok ? r.arrayBuffer() : Promise.reject())
      .then(buf => new FontFace(fam, buf, { weight }).load()).then(f => document.fonts.add(f)).catch(() => {});
  });
}

// ---------------------------------------------------------------- 入场动画 ---
function isStandalone(){
  try { return (window.matchMedia && matchMedia('(display-mode: standalone)').matches) || navigator.standalone === true; } catch(e){ return false; }
}
function isPhone(){
  const ua = navigator.userAgent || '';
  if (/Android|iPhone|iPod/.test(ua)) return true;
  try { return matchMedia('(pointer: coarse)').matches && Math.min(screen.width, screen.height) < 600; } catch(e){ return false; }
}
function shouldSplash(){
  if (!isStandalone() || !isPhone()) return false;
  try { if (sessionStorage.getItem('nyra-splash-shown')) return false; sessionStorage.setItem('nyra-splash-shown', '1'); } catch(e){}
  return true;
}
function setThemeColor(c){ document.querySelectorAll('meta[name=theme-color]').forEach(m => { if (!m.dataset.orig) m.dataset.orig = m.content; m.content = c || m.dataset.orig; }); }
function runSplash(){
  const s = $('splash');
  if (!shouldSplash()) { s.remove(); return; }
  let reduce = false; try { reduce = matchMedia('(prefers-reduced-motion: reduce)').matches; } catch(e){}
  s.classList.remove('hidden'); s.dataset.state = 'playing'; setThemeColor('#0B0B0C');
  const finish = () => {
    s.dataset.state = 'done'; s.classList.add('out'); setThemeColor('');
    setTimeout(() => s.remove(), 480);
  };
  if (reduce) { s.classList.add('reduce'); s.innerHTML = splashSvg(SPLASH_END); setTimeout(finish, 1400); return; }
  s.classList.add('play');
  let start = null;
  const step = (ts) => {
    if (start === null) start = ts;
    const sec = (ts - start) / 1000;
    s.innerHTML = splashSvg(Math.min(sec, SPLASH_END));
    s.dataset.t = sec.toFixed(2);
    if (sec >= SPLASH_HOLD) { finish(); return; }
    if (sec >= SPLASH_END) { setTimeout(() => finish(), (SPLASH_HOLD - sec) * 1000); return; }
    requestAnimationFrame(step);
  };
  s.innerHTML = splashSvg(0);
  requestAnimationFrame(step);
}
